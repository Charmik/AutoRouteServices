#nging config for https://testing-routecycle.duckdns.org testing stage
#file is located here: /etc/nginx/sites-available/routecycle.test
#to connect to server: ssh root@$do_testing

# Same prefix layout as production (frontend/routecycle.com), so a web build that works here works
# there. Staging runs ONE backend (hz2) for every class of traffic, so all four upstreams point at
# it -- the names exist to keep this file diffable against the prod one and to make re-pointing a
# single class a one-line edit.
#
# hz2 = 88.99.161.250 (see docs/ARCHITECTURE.md).
upstream routecycle_test_api    { server 88.99.161.250:7070; }
# All Strava paths. To move Strava to another box, change only this line.
upstream routecycle_test_strava { server 88.99.161.250:7070; }
upstream routecycle_test_road   { server 88.99.161.250:7070; }
upstream routecycle_test_gravel { server 88.99.161.250:7070; }

server {
    server_name testing-routecycle.duckdns.org www.testing-routecycle.duckdns.org;

    #root /home/charm/routeplanner/dist;
    root /home/panda/routeplanner/www;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # Strava, carved out of /api/v1 so it can be re-pointed on its own. nginx picks the LONGEST
    # matching prefix, so these three win over /api/v1 below regardless of order in this file.
    location /api/v1/strava {
        proxy_pass http://routecycle_test_strava/api/v1/strava;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    location /api/v1/auth/strava {
        proxy_pass http://routecycle_test_strava/api/v1/auth/strava;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    location /api/v1/activities {
        proxy_pass http://routecycle_test_strava/api/v1/activities;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    # Road generation. The trailing /api/v1 replaces the matched /api/road/v1 prefix, so the backend
    # -- which registers only /api/v1/* -- sees the path it expects.
    location /api/road/v1 {
        proxy_pass http://routecycle_test_road/api/v1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    # Gravel generation. On staging the same backend serves both graphs and picks by the request's
    # `profile` field, so this points at hz2 like everything else.
    location /api/gravel/v1 {
        proxy_pass http://routecycle_test_gravel/api/v1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    # The profile-neutral API: everything user-scoped, plus legacy un-prefixed clients.
    # NOT legacy and NOT safe to delete -- the web app's VITE_BACKEND_API_URL points here.
    location /api/v1 {
        proxy_pass http://routecycle_test_api/api/v1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    location /reverse {
        proxy_pass http://176.9.23.116:2322/reverse;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }

    # Photon geocoder, and the CATCH-ALL for /api: any /api/... prefix not declared above lands here
    # and gets Photon's own Javalin 404, which reads like an AutoRoute bug but is not.
    location /api {
        proxy_pass http://176.9.23.116:2322/api;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass_request_headers on;
    }



    # Optional: serve static files with caching
    location ~* \.(jpg|jpeg|png|gif|ico|css|js)$ {
        expires 365d;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/testing-routecycle.duckdns.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/testing-routecycle.duckdns.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}

server {
    if ($host = testing-routecycle.duckdns.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80 default_server;
    server_name testing-routecycle.duckdns.org www.testing-routecycle.duckdns.org;
    return 301 https://$server_name$request_uri;


}
